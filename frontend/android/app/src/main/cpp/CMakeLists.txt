# ARM64 CPU inference with the R2T2-tested llama.cpp revision.
cmake_minimum_required(VERSION 3.22)
project(resonance_r2t2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
if(ANDROID)
    # Optimize C kernels too, while keeping JNI and capability scoring on the ARM64 baseline.
    add_compile_options(-O3)
    add_link_options("-Wl,-z,max-page-size=16384")
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ARM_ARCH "" CACHE STRING "" FORCE)
    # AGP packages modules from its ABI output directory, not the upstream executable directory.
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_LIBRARY_OUTPUT_DIRECTORY})
endif()
add_subdirectory(../../../../third_party/llama.cpp llama)
if(ANDROID)
    # Android loads unversioned SONAMEs from its extracted native library directory.
    foreach(native_lib ggml ggml-base llama mtmd)
        set_target_properties(${native_lib} PROPERTIES VERSION "" SOVERSION "")
    endforeach()
    add_library(resonance-r2t2 SHARED r2t2_engine.cpp r2t2_jni.cpp r2t2_backend.cpp)
    target_link_libraries(resonance-r2t2 PRIVATE llama mtmd android log dl)
    target_link_options(resonance-r2t2 PRIVATE "-Wl,-z,max-page-size=16384")
else()
    # Same engine and prompt on a host, for reproducible sample validation.
    add_executable(resonance-r2t2-test r2t2_engine.cpp r2t2_test.cpp)
    target_link_libraries(resonance-r2t2-test PRIVATE llama mtmd)
endif()
